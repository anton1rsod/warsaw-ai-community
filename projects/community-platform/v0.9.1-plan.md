Apply to: `ProfileEditor`, `OnboardForm`, `EventForm`, `InviteForm`, `InviteUrlDisplay`. Markdown previews use **`.prose-warm`** (C2), not `prose prose-neutral`.

### C8. Modal recipe (native `<dialog>` — `ConsentModal`)

Use the native HTML `<dialog>` opened with `.showModal()` (2026 a11y best practice — top-layer, `inert` background, Escape, `::backdrop`; **no manual focus trap** per the WAI-APA consensus).

```
element:  <dialog ref> — ref.showModal() in a mount effect; Escape fires onCancel via the `cancel` event
scrim:    backdrop:bg-ink/50                 (Tailwind `backdrop:` variant → dialog::backdrop)
card:     bg-cream border border-ink p-6 max-w-md text-ink   (zero radius; the <dialog> IS the card)
title:    font-display font-semibold text-ink + id → aria-labelledby
body:     font-body text-ink / font-voice text-dust
buttons:  <Pill variant="solid">accept</Pill>  <Pill variant="dashed">cancel</Pill>
```

### C9. Metric-tile + table recipe (`/admin/health`)

```
tile:     bg-paper border-l-[3px] border-l-ink p-4
number:   font-voice text-[40px] tabular-nums text-ink
sublabel: font-voice text-[11px] text-dust
table:    th → font-voice text-[11px] uppercase tracking-[1px] text-dust text-left py-2
          td → font-voice text-ink (or text-dust for ratios) tabular-nums py-2
          row separator → border-b border-ink/15      (NOT `border-b` default gray; NOT `rounded border`)
```

### C-E2E. E2E-mode mock pattern (Phase 4 — mirror `app/actions/_test-status-store.ts` + `app/actions/status.ts`)

- Store lives on `globalThis` (Next 16 `"use server"` files are a separate module graph — without the hop, action-writes and page-reads see different instances).
- Each store exports its mock-actions object + `isE2EMode()` (`process.env.NEXT_PUBLIC_E2E_MODE === "1"`) + `reset()`.
- Each action forks **after Zod parse, before auth/GitHub**: `if (!isProductionRuntime() && isE2EMode()) return fromMock(await mock…Actions.…(parsed.data));` where `isProductionRuntime()` = `process.env.NODE_ENV === "production"` (the prod double-guard).
- Stores are coverage-excluded in `vitest.config.ts` (E2E-only — no Vitest reaches them).

```tsx
"use client";

import { useEffect, useRef } from "react";
import { Pill } from "@/app/components/Pill";

export interface ConsentModalProps {
  onAccept: () => void;
  onCancel: () => void;
  disabled: boolean;
}

export function ConsentModal({ onAccept, onCancel, disabled }: ConsentModalProps): React.JSX.Element {
  const ref = useRef<HTMLDialogElement>(null);

  // showModal() puts the dialog in the top layer + makes the background inert
  // + enables Escape + ::backdrop. No manual focus trap (WAI-APA consensus).
  useEffect(() => {
    const dialog = ref.current;
    if (dialog && !dialog.open) dialog.showModal();
  }, []);

  return (
    <dialog
      ref={ref}
      aria-labelledby="consent-modal-title"
      onCancel={(e) => {
        e.preventDefault();   // Escape → route through onCancel (sign out)
        if (!disabled) onCancel();
      }}
      className="backdrop:bg-ink/50 bg-cream border border-ink p-6 max-w-md text-ink"
    >
      <h2 id="consent-modal-title" className="font-display font-semibold text-ink text-xl">
        Opt in to the Warsaw AI Community platform
      </h2>
      <ul className="mt-4 space-y-2 font-body text-ink text-sm">
        <li>Your status updates will be committed to a public MIT-licensed repository.</li>
        <li>You can edit or delete them at any time.</li>
        <li>You can request full data export or deletion of your profile at any time.</li>
      </ul>
      <div className="mt-6 flex gap-3">
        <Pill variant="solid" type="button" onClick={onAccept} disabled={disabled}>
          Accept and continue
        </Pill>
        <Pill variant="dashed" type="button" onClick={onCancel} disabled={disabled}>
          Cancel — go back
        </Pill>
      </div>
    </dialog>
  );
}
```

/**
 * In-memory mock store for RSVP E2E tests. Active only when
 * NEXT_PUBLIC_E2E_MODE === "1". Backs both rsvp-event.ts (write) and the
 * /api/event-rsvp-state hydration read so the toggle is deterministic.
 * Coverage-excluded in vitest.config.ts — E2E-only.

import { auth } from "@/lib/auth";
import { findMemberByHandle } from "@/lib/content-snapshot";

interface MockRsvpShared {
  // key = `${memberSlug}:${eventSlug}` → state; sha is per member-profile
  state: Map<string, "going" | "interested" | "none">;
  sha: Map<string, string>; // memberSlug → current profileSha
  counter: number;
}
const G = globalThis as { __waicMockRsvpStore?: MockRsvpShared };
function shared(): MockRsvpShared {
  if (!G.__waicMockRsvpStore) {
    G.__waicMockRsvpStore = { state: new Map(), sha: new Map(), counter: 0 };
  }
  return G.__waicMockRsvpStore;
}
function nextSha(): string { const s = shared(); s.counter += 1; return `mockrsvpsha-${s.counter}`; }
function curSha(slug: string): string {
  const s = shared();
  if (!s.sha.has(slug)) s.sha.set(slug, nextSha());
  return s.sha.get(slug) as string;
}
async function resolveAuthState(): Promise<
  { ok: true; slug: string } | { ok: false; error: "not_authenticated" | "not_a_member" }
> {
  const session = await auth();
  if (!session?.githubHandle) return { ok: false, error: "not_authenticated" };
  const member = findMemberByHandle(session.githubHandle);
  if (!member) return { ok: false, error: "not_a_member" };
  return { ok: true, slug: member.slug };
}

export type MockRsvpResult =
  | { ok: true; state: "going" | "interested" | "none" }
  | { ok: false; error: string };

export const mockRsvpActions = {
  /** Mirrors rsvpEvent(): SHA-gate then set state. */
  async toggle(input: { eventSlug: string; desiredState: "going" | "interested" | "none"; profileSha: string }): Promise<MockRsvpResult> {
    const a = await resolveAuthState();
    if (!a.ok) return { ok: false, error: a.error };
    if (input.profileSha !== curSha(a.slug)) return { ok: false, error: "refresh_needed" };
    shared().state.set(`${a.slug}:${input.eventSlug}`, input.desiredState);
    shared().sha.set(a.slug, nextSha()); // write rotates the sha
    return { ok: true, state: input.desiredState };
  },
  /** Mirrors the /api/event-rsvp-state read: current state + profileSha for the viewer. */
  async getState(eventSlug: string): Promise<{ state: "going" | "interested" | "none"; profileSha: string } | null> {
    const a = await resolveAuthState();
    if (!a.ok) return null;
    return { state: shared().state.get(`${a.slug}:${eventSlug}`) ?? "none", profileSha: curSha(a.slug) };
  },
  reset(): void { const s = shared(); s.state.clear(); s.sha.clear(); s.counter = 0; },
};

export function isE2EMode(): boolean { return process.env.NEXT_PUBLIC_E2E_MODE === "1"; }
